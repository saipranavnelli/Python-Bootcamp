class Customer:
    bname="XYZ Bank"
    ifsc="XYZ00001234"
    
    def __init__(self,acno,name,pin,bal=0):
        self.cacno=acno
        self.cname=name
        self.pin=pin
        self.bal=bal
        
    def __str__(self):
        return f"AcNo: {self.cacno}\nName: {self.cname}\nBalance: {self.bal}"

    def deposit(self,pin,damt):
        if pin==self.pin:
            if damt>0:
                self.bal+=damt
                print("Deposited Successfully")
                print(f"your current balance is: {self.bal}")
            else:
                print("Invalid Deposited Ammount")
        else:
            print("Invalid Pin")

    def withdraw(self,pin,wamt):
        if pin==self.pin:
            if wamt>0:
                if wamt<=self.bal:
                    self.bal-=wamt
                    print("Withdraw Successfully")
                    print(f"your current balance is: {self.bal}")
                else:
                    print("Insufficient Funds in Your Account")
            else:
                print("Invalid Withdraw ammount")
        else:
            print("Invalid Pin")
                    
    def cust_info(self):
        print(f"Customer Name: {self.cname}")
        print(f"Customer AcNo: {self.cacno}")
        print(f"Customer Balance: {self.bal}")
        print(f"Bankname: {Customer.bname}")
        print(f"IFSC Code: {Customer.ifsc}")

class Calculator:
    
    def add(self):
        x,y=eval(input("enter your input's: "))
        res=x+y
        return f"the addition of {x} and {y} is: {res}"
    
    def sub(self):
        x,y=eval(input("enter your input's: "))
        res=x-y
        return f"the subtraction of {x} and {y} is: {res}"
    
    def mul(self):
        x,y=eval(input("enter your input's: "))
        res=x*y
        return f"the multiplication of {x} and {y} is: {res}"
    
    def div(self):
        x,y=eval(input("enter your input's: "))
        if y==0:
            return "sno can't be zero"
        else:
            res=x/y
        return f"the division of {x} and {y} is: {res:.2f}"

c=Calculator()
while True:
    print("Select any one option from the following options:\
\n\t 1).Addition\n\t 2).Subtraction\n\t 3).Multiplication\
\n\t 4).Division\n\t 5).Exit")
    opt=int(input("enter your option: "))
    if opt==1:
        print(c.add())
    elif opt==2:
        print(c.sub())
    elif opt==3:
        print(c.mul())
    elif opt==4:
        print(c.div())
    elif opt==5:
        break
    else:
        print("Invalid Option")
        
print("Thanks for using my application")
